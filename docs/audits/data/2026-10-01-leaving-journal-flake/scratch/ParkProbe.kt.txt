object Park {
    @Volatile var armed = false
    val parked = java.util.concurrent.CountDownLatch(1)      // main counts down when it is inside measure
    val resume = java.util.concurrent.CountDownLatch(1)      // helper counts down to let main leave measure
    val tick = androidx.compose.runtime.mutableStateOf(0)    // read in measure, so a write dirties this node's measurement
}

/** A layout modifier whose measure reads Park.tick and, when armed, parks the calling thread on Park.resume. */
fun Modifier.parkInMeasure(): Modifier = this.layout { measurable, constraints ->
    Park.tick.value                                          // the state read that makes a write remeasure this node
    if (Park.armed) {
        Park.armed = false
        ThreadProbe.record("PARK-ENTER")                     // logs thread, main?, isDeferringContinuations
        Park.parked.countDown()
        Park.resume.await(5, java.util.concurrent.TimeUnit.SECONDS)
        ThreadProbe.record("PARK-EXIT")
    }
    val p = measurable.measure(constraints)
    layout(p.width, p.height) { p.place(0, 0) }
}
